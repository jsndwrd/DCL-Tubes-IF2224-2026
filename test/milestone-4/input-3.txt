program ArithmeticTest;
var
  a, b, c: integer;

begin
  a := 5;
  b := 10;
  c := a + b;
  writeln(c);
  c := b - a;
  writeln(c);
  c := a * b;
  writeln(c);
  c := b div a;
  writeln(c);
  if a < b then
    begin
      writeln(a);
    end;
program ForRepeat;

var
  i, sum: integer;

begin
  sum := 0;
  for i := 1 to 5 do
  begin
    sum := sum + i;
  end;
  writeln(sum);

  repeat
    sum := sum - 1;
  until sum == 0;
  writeln(sum);
end.
